    private void updatePronunciation(String result, LanguageProfile language) {
        if (pronunciationView == null) return;
        String guide = koreanPronunciation(result, language);
        if (guide == null || guide.trim().isEmpty()) {
            pronunciationView.setText("");
            pronunciationView.setVisibility(View.GONE);
            return;
        }
        pronunciationView.setText("한국어 발음(참고)  " + guide);
        pronunciationView.setVisibility(View.VISIBLE);
    }

    private String koreanPronunciation(String text, LanguageProfile language) {
        if (language == null || text == null || text.trim().isEmpty() || "ko".equals(language.id)) return "";
        if ("vi".equals(language.id)) return vietnamesePronunciation(text);
        if ("en".equals(language.id)) return englishPronunciation(text);
        return "";
    }

    private String pronounceWords(String text, boolean vietnamese) {
        String[] words = text.trim().split("\\s+");
        StringBuilder out = new StringBuilder();
        for (String token : words) {
            if (token.isEmpty()) continue;
            String punctuation = token.endsWith("?") ? "?" : token.endsWith("!") ? "!" : token.endsWith(".") ? "." : token.endsWith(",") ? "," : "";
            String core = token.replaceAll("[^\\p{L}']", "");
            if (core.isEmpty()) continue;
            String spoken = vietnamese ? pronounceVietnameseWord(core) : pronounceEnglishWord(core);
            if (spoken == null || spoken.isEmpty()) continue;
            if (out.length() > 0) out.append(' ');
            out.append(spoken).append(punctuation);
        }
        return out.toString();
    }
